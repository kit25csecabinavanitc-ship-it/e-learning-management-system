Database Schema & Entity-Relationship Architecture

Engineered in compliance with Third Normal Form (3NF) relational design standards. This page provides an interactive view of all 10 entities, relational cardinalities, primary/foreign key mappings, and sample SQL definitions.

Core Relational Business Rules & Cardinalities

1. Students ↔ Courses (M:N)

One student can enroll in multiple courses, and each course enrolls multiple students. Resolved via associative junction table enrollments.

2. Courses ↔ Lessons (1:N)

Each course entity is composed of multiple sequential lessons. lessons.course_id establishes the foreign key relationship.

3. Students ↔ Lessons (Progress)

Connected via lesson_progress to record individual lesson completion flags and student notes.

4. Quizzes ↔ Quiz Attempts (1:N)

One student can attempt an assessment multiple times. Each attempt records scores, pass status, and completion timestamp.

-- =======================================================================
-- LearnSphere DBMS - Production MySQL Relational Table Definitions
-- 3rd Normal Form (3NF) Architecture with Referential Integrity Constraints
-- =======================================================================

CREATE DATABASE IF NOT EXISTS learnsphere_db
CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

USE learnsphere_db;

-- 1. Students Entity
CREATE TABLE IF NOT EXISTS students (
    student_id VARCHAR(36) PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    email VARCHAR(120) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    avatar VARCHAR(10) DEFAULT 'ST',
    bio TEXT,
    enrolled_date DATE NOT NULL,
    status ENUM('active', 'suspended', 'graduated') DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- 2. Courses Entity
CREATE TABLE IF NOT EXISTS courses (
    course_id VARCHAR(36) PRIMARY KEY,
    title VARCHAR(150) NOT NULL,
    instructor VARCHAR(100) NOT NULL,
    category VARCHAR(50) NOT NULL,
    difficulty_level ENUM('Beginner', 'Intermediate', 'Advanced') NOT NULL,
    duration_hours VARCHAR(30) NOT NULL,
    price VARCHAR(20) DEFAULT 'Free',
    description TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- 3. Enrollments Associative Junction Table (Resolves M:N Cardinality)
CREATE TABLE IF NOT EXISTS enrollments (
    enrollment_id VARCHAR(36) PRIMARY KEY,
    student_id VARCHAR(36) NOT NULL,
    course_id VARCHAR(36) NOT NULL,
    enrolled_at DATE NOT NULL,
    status ENUM('in-progress', 'completed', 'dropped') DEFAULT 'in-progress',
    progress_percent INT DEFAULT 0 CHECK (progress_percent BETWEEN 0 AND 100),
    CONSTRAINT fk_enrollment_student 
        FOREIGN KEY (student_id) REFERENCES students(student_id) 
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_enrollment_course 
        FOREIGN KEY (course_id) REFERENCES courses(course_id) 
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT uq_student_course UNIQUE (student_id, course_id)
) ENGINE=InnoDB;

-- 4. Lessons Entity (1:N with Courses)
CREATE TABLE IF NOT EXISTS lessons (
    lesson_id VARCHAR(36) PRIMARY KEY,
    course_id VARCHAR(36) NOT NULL,
    module_name VARCHAR(100) NOT NULL,
    title VARCHAR(150) NOT NULL,
    duration VARCHAR(30) NOT NULL,
    video_url VARCHAR(255),
    content_text LONGTEXT NOT NULL,
    sequence_order INT NOT NULL DEFAULT 1,
    CONSTRAINT fk_lesson_course 
        FOREIGN KEY (course_id) REFERENCES courses(course_id) 
        ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB;
